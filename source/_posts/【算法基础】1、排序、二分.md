==快速排序算法模板==
```cpp
void quick_sort(int q[], int l, int r)
{
    if (l >= r) return;

    int i = l - 1, j = r + 1, x = q[l + r >> 1];
    while (i < j)
    {
        do i ++ ; while (q[i] < x);
        do j -- ; while (q[j] > x);
        if (i < j) swap(q[i], q[j]);
    }
    quick_sort(q, l, j);
    quick_sort(q, j + 1, r);
}
```

需要注意的是如果以i为边界：
```cpp

        do j -- ; while (q[j] > x);
        if (i < j) swap(q[i], q[j]);
    }
    quick_sort(q, l, j); //如果 quick_sort(q, l, i -  1)  j
    quick_sort(q, j + 1, r); //如果 quick_sort(q, i, r)  j + 1 一定不能取到q[r] 
    // 样例 1， 2  若 x =  1
    // [0,1]区间会死循环 所以一定不可以取到q[l]
}
```

`C++代码`
```cpp

const int N = 1e6 + 10;

int q[N];


//l是左指针 r是右指针 左大于等于右了 就相遇了
void quick_sort(int q[], int l, int r) {
    if (l >= r) return;  // 区间中没有数字 或者只有一个数字

    int i = l - 1, j = r + 1, x = q[l + r >> 1]; // 取中点 要注意此处 为什么是i = l - 1, j = r + 1
    while (i < j) {
        do i++; while (q[i] < x);
        do j--; while (q[j] > x);
        if (i < j) swap(q[i], q[j]);
    }

    quick_sort(q, l, j);
    quick_sort(q, j + 1, r);
}
```

```python
def quick_sort(q, l, r):
    if l >= r:
        return
    i = l - 1
    j = r + 1
    x = q[(l+r) >> 1]
    while i < j:
        i+=1
        while q[i] < x:
            i += 1
        j-=1
        while q[j] > x:
            j -= 1
        if i < j:
            q[i], q[j] = q[j], q[i]

    quick_sort(q, l, j)
    quick_sort(q, j+1, r)
```

>归并排序是先分 在递归处理两边

==归并排序算法模板==
```cpp
void merge_sort(int q[], int l, int r)
{
    if (l >= r) return;

    int mid = l + r >> 1;
    merge_sort(q, l, mid);
    merge_sort(q, mid + 1, r);

    int k = 0, i = l, j = mid + 1;
    while (i <= mid && j <= r)
        if (q[i] <= q[j]) tmp[k ++ ] = q[i ++ ];
        else tmp[k ++ ] = q[j ++ ];

    while (i <= mid) tmp[k ++ ] = q[i ++ ];
    while (j <= r) tmp[k ++ ] = q[j ++ ];

    for (i = l, j = 0; i <= r; i ++, j ++ ) q[i] = tmp[j];
}
```

```cpp
#include <iostream>

using namespace std;

const int  N = 1e5 + 10;

int n;

void merge_sort(int q[], int l, int r)
{
    if(l >= r) return;
    
    int mid = l + r >> 1;
    merge_sort(q, l, mid), merge_sort(q, mid + 1, r);
    
    int k = 0, i = l, j = mid + 1;//k是代表 temp中已经有多少个数 即已经合并多少个数了

        if(q[i] <= q[j]) temp[k ++ ] = q[i ++ ];
        else temp[k ++ ] = q[j ++ ];
    }
    while(i <= mid ) temp[k ++ ] = q[i ++ ];//左半边没有循环完
    while(j <= r )   temp[k ++ ] = q[j ++ ];
    
    for(i = l,j = 0; i <= r; i ++,j ++ ) q[i] = temp[j];
}
```

# 2 二分
>有单调性 一定可以二分，但是可以二分的题目不一定非要有单调性
>二分的本质是只要在区间中可以找到一种性质，那么就可以使用二分把边界点二分出来(可以把区间一分为二)。

>有单调性的题目可以二分，但是可以二分的题目不一定要有单调性，所以二分的本质并不是单调性。
>如果可以找到一个性质，把整个区间一分为二，一半满足一半不满足，二分就可以寻找这样性质的边界。

==整数二分算法模板==
```cpp
bool check(int x) {/* ... */} // 检查x是否满足某种性质

// 区间[l, r]被划分成[l, mid]和[mid + 1, r]时使用：
int bsearch_1(int l, int r)
{
    while (l < r)
    {
        int mid = l + r >> 1;
        if (check(mid)) r = mid;    // check()判断mid是否满足性质
        else l = mid + 1;
    }
    return l;
}
// 区间[l, r]被划分成[l, mid - 1]和[mid, r]时使用：
int bsearch_2(int l, int r)
{
    while (l < r)
    {
        int mid = l + r + 1 >> 1;
        if (check(mid)) l = mid;
        else r = mid - 1;
    }
    return l;
}
```
>二分的结果一定有解。
## 2.2 浮点数二分
==浮点数二分算法模板1==
```cpp
#include <iostream>

using namespace std;

int main()
{
    double x;
    cin >> x;
    
    double l = -x, r = x;
    while(r - l > 1e-8) //也可for迭代100次
    {
        double mid = (l + r) / 2 ;
        if(mid * mid  >= x ) r = mid;
        else l = mid;
    }
    //如果题目要求保留4位小数 r - l > 1e-6
    //如果题目要求保留5位小数 r - l > 1e-5
    //如果题目要求保留6位小数 r - l > 1e-8
    printf("%lf\n", l);
}
```
